    ```ts
    import { test, expect } from '@playwright/test';

    // VER-01 — real end-to-end purchase. REAL MONEY: Celitech provisions a real
    // low-cost eSIM. MANUAL RUN ONLY — never in `npm test` or CI.
    // Requires the app running with NEXT_PUBLIC_STRIPE_MOCK=false and live keys
    // (mock mode skips the webhook -> no real Celitech call, no email).
    test('completes a real purchase and provisions an eSIM', async ({ page }) => {
      // 1. Browse — land on the catalog.
      await page.goto('/en/browse');
      await expect(page.getByRole('heading', { level: 1 })).toBeVisible();

      // 2. Pick a destination, then the cheapest plan, then go to checkout.
      //    Confirm these selectors against destination-card.tsx + esim/[slug] plan list.
      await page.getByRole('link', { name: /.../ }).first().click();   // a destination
      // ... select the cheapest plan -> navigates to /en/checkout?plan=<realPlanId>

      // 3. Fill the email field.
      await page.getByRole('textbox', { name: /email/i }).fill('e2e+ver01@esimpanda.co');

      // 4. Stripe PaymentElement is a cross-origin iframe — use frameLocator.
      //    Selector may need 1-2 tuning passes (Stripe DOM is @stripe/stripe-js@9.2.0
      //    version-sensitive — Open Question 1). Inspect the live DOM with --debug.
      const stripeFrame = page.frameLocator('iframe[title*="payment" i], iframe[name^="__privateStripeFrame"]');
      await expect(stripeFrame.locator('[name="number"]')).toBeVisible();   // Pitfall 5: await iframe
      await stripeFrame.locator('[name="number"]').fill('4242 4242 4242 4242');
      await stripeFrame.locator('[name="expiry"]').fill('12 / 30');
      await stripeFrame.locator('[name="cvc"]').fill('123');
      // PaymentElement may also need a postal code / country (Open Question 2) —
      // inspect the rendered Element and fill any required field.

      // 5. Pay.
      await page.getByRole('button', { name: /pay/i }).click();

      // 6. Success page — provisioning is async (delivery-page polls up to ~60s).
      await expect(page).toHaveURL(/\/checkout\/success\?payment_intent=pi_/, { timeout: 30_000 });
      await expect(page.getByRole('heading', { name: /your esim is ready/i }))
        .toBeVisible({ timeout: 90_000 });   // Pitfall 6: generous timeout for real Celitech round-trip

      // 7. QR present — qr-code-display.tsx renders a <QRCodeSVG> (<svg>).
      await expect(page.locator('svg').first()).toBeVisible();
    });
    ```

    - Do NOT assert on the Resend inbox — email delivery is verified by the provisioning code path completing without error; the inbox check is a documented manual companion step.
