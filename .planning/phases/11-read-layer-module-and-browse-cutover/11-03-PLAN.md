Migrated store shape — implement src/stores/comparison.ts exactly as:
```ts
import { create } from 'zustand';
import type { Plan } from '@/lib/db/destinations';

interface ComparisonState {
  selectedPlans: Plan[];                   // was: selectedPlanIds: string[]
  isSheetOpen: boolean;
  togglePlan: (plan: Plan) => void;        // was: (id: string)
  clearSelection: () => void;
  openSheet: () => void;
  closeSheet: () => void;
}

export const useComparisonStore = create<ComparisonState>((set) => ({
  selectedPlans: [],
  isSheetOpen: false,
  togglePlan: (plan) =>
    set((state) => {
      if (state.selectedPlans.some((p) => p.id === plan.id)) {
        return { selectedPlans: state.selectedPlans.filter((p) => p.id !== plan.id) };
      }
      if (state.selectedPlans.length >= 3) return state;
      return { selectedPlans: [...state.selectedPlans, plan] };
    }),
  clearSelection: () => set({ selectedPlans: [], isSheetOpen: false }),
  openSheet: () => set({ isSheetOpen: true }),
  closeSheet: () => set({ isSheetOpen: false }),
}));
```
Note: `comparison.ts` uses a plain `create()` — NO `persist` middleware. Confirmed by RESEARCH (read the current file). Therefore NO version bump / migrate function is needed.
